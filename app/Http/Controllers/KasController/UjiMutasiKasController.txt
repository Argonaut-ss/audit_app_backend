<?php

namespace App\Http\Controllers\KasController;

use App\Http\Controllers\Controller;
use App\Models\JwbKasus;
use App\Models\Kas\CashCountKas;
use App\Models\Kas\Kas;
use App\Models\Kas\UjiMutasiKas;
use App\Models\Kas\RekapMutasiKas;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class UjiMutasiKasController extends Controller
{
    private function authorizedSources(Request $request, int $kasId): array
    {
        $kas = Kas::findOrFail($kasId);
        JwbKasus::forUser($request->user())
            ->where('JwbKasusID', $kas->JwbKasusID)
            ->firstOrFail();

        $cashCount = CashCountKas::where('KasID', $kas->KasID)->firstOrFail();
        $rekapMutasi = RekapMutasiKas::where('KasID', $kas->KasID)
            ->orderByDesc('RekapMutasiID')
            ->firstOrFail();

        return [$kas, $cashCount, $rekapMutasi];
    }

    private function calculate(CashCountKas $cashCount, RekapMutasiKas $rekapMutasi): array
    {
        $hasilCashCount = (int) $cashCount->TotalKeseluruhan;
        $kasMasuk = (int) $rekapMutasi->DebitTotal;
        $kasKeluar = (int) $rekapMutasi->KreditTotal;
        $saldoBuku = (int) $rekapMutasi->SaldoAwal;
        $saldoAuditSblm = $hasilCashCount + $kasMasuk - $kasKeluar;
        $saldoAuditStlh = $kasKeluar - $kasMasuk + $hasilCashCount;

        return [
            'HasilCashCount' => $hasilCashCount,
            'KasKeluar' => $kasKeluar,
            'KasMasuk' => $kasMasuk,
            'SaldoBuku' => $saldoBuku,
            'SaldoAuditSblm' => $saldoAuditSblm,
            'SaldoAuditStlh' => $saldoAuditStlh,
            'SelisihSblm' => $saldoAuditSblm - $saldoBuku,
            'SelisihStlh' => $saldoAuditStlh - $saldoBuku,
        ];
    }

    private function serialize(UjiMutasiKas $ujiMutasi, array $calculated): array
    {
        return [
            'UjiMutasiID' => $ujiMutasi->UjiMutasiID,
            'KasID' => $ujiMutasi->KasID,
            'CashCountID' => $ujiMutasi->CashCountID,
            'RekapMutasiID' => $ujiMutasi->RekapMutasiID,
            'HasilCashCount' => $calculated['HasilCashCount'],
            'KasKeluar' => $calculated['KasKeluar'],
            'KasMasuk' => $calculated['KasMasuk'],
            'SaldoBuku' => $calculated['SaldoBuku'],
            'SaldoAuditSblm' => $ujiMutasi->SaldoAuditSblm,
            'SaldoAuditStlh' => $ujiMutasi->SaldoAuditStlh,
            'SelisihSblm' => $ujiMutasi->SelisihSblm,
            'SelisihStlh' => $ujiMutasi->SelisihStlh,
            'Penjelasan' => $ujiMutasi->Penjelasan,
            'created_at' => $ujiMutasi->created_at,
            'updated_at' => $ujiMutasi->updated_at,
        ];
    }

    public function show(Request $request, int $kasId): JsonResponse
    {
        [$kas, $cashCount, $rekapMutasi] = $this->authorizedSources($request, $kasId);
        $calculated = $this->calculate($cashCount, $rekapMutasi);
        $ujiMutasi = UjiMutasiKas::where('KasID', $kas->KasID)->first();

        return response()->json([
            'success' => true,
            'data' => [
                'UjiMutasiID' => $ujiMutasi?->UjiMutasiID,
                'KasID' => $kas->KasID,
                'CashCountID' => $cashCount->CashCountID,
                'RekapMutasiID' => $rekapMutasi->RekapMutasiID,
                'HasilCashCount' => $calculated['HasilCashCount'],
                'KasKeluar' => $calculated['KasKeluar'],
                'KasMasuk' => $calculated['KasMasuk'],
                'SaldoBuku' => $calculated['SaldoBuku'],
                'SaldoAuditSblm' => $calculated['SaldoAuditSblm'],
                'SaldoAuditStlh' => $calculated['SaldoAuditStlh'],
                'SelisihSblm' => $calculated['SelisihSblm'],
                'SelisihStlh' => $calculated['SelisihStlh'],
                'Penjelasan' => $ujiMutasi?->Penjelasan,
            ],
        ]);
    }

    public function store(Request $request, int $kasId): JsonResponse
    {
        $validated = $request->validate([
            'Penjelasan' => ['nullable', 'string', 'max:255'],
        ]);
        [$kas, $cashCount, $rekapMutasi] = $this->authorizedSources($request, $kasId);
        $calculated = $this->calculate($cashCount, $rekapMutasi);

        $ujiMutasi = DB::transaction(function () use (
            $kas,
            $cashCount,
            $rekapMutasi,
            $validated,
            $calculated
        ) {
            return UjiMutasiKas::updateOrCreate(
                ['KasID' => $kas->KasID],
                [
                    'CashCountID' => $cashCount->CashCountID,
                    'RekapMutasiID' => $rekapMutasi->RekapMutasiID,
                    'SaldoAuditSblm' => $calculated['SaldoAuditSblm'],
                    'SaldoAuditStlh' => $calculated['SaldoAuditStlh'],
                    'SelisihSblm' => $calculated['SelisihSblm'],
                    'SelisihStlh' => $calculated['SelisihStlh'],
                    'Penjelasan' => $validated['Penjelasan'] ?? null,
                ]
            );
        });

        return response()->json([
            'success' => true,
            'message' => 'Uji mutasi kas berhasil disimpan.',
            'data' => $this->serialize($ujiMutasi, $calculated),
        ]);
    }
}
